// ===== General =====
?hp < 10
 activate potion

?hp < 10 & item.potion = empty
 heal = 1

?hp > 40
 heal = 0

?heal = 1
 equipL Crusher
 equipR poison shield
:?foe = boss | foe.count > 5
 equip blade
:
 equipL Crusher
 equipR poison shield

?item.CanActivate("blade")
 activate R
?item.CanActivate("heavy_hammer") &
^poena = 0
 activate R

// ===== Ghosts: fire wand =====
?ghost = 1
 equipL fire wand
 equipR poison shield

// ===== Summon abilities =====
?hasWisp = 1 & foe.count > 0 &
^summon.GetVar("ignition") > 2 &
^item.GetCooldown("cinderwisp") <= 0 &
^item.CanActivate()
 activate cinderwisp

?hasVoid = 1 & foe.count > 0 &
^item.GetCooldown("voidweaver") <= 0 &
^item.CanActivate()
 activate voidweaver

// ===== Temple: tongue dodge =====
?tongue = 1 & wrap = 0 &
^foe.time >= nagTime &
^item.GetCooldown("mind") <= 0
 equipL mind

// ===== Temple: wrapped, blade only =====
?wrap = 1
 equip blade
