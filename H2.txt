// ===== Plateau: phase =====
ph = 0
?loc = rocky & heal = 0 &
^foe.id = dysangelos_elementalist
 ph = 2
?loc = rocky & heal = 0 &
^foe.id = dysangelos_perfected
 ph = 3

// Phase 2: counter element
?ph = 2 & foe = poison
 equipL ice sword
 equipR ice shield
?ph = 2 & foe = vigor
 equipL poison sword
 equipR poison shield
?ph = 2 & foe = aether
 equipL vigor sword
 equipR vigor shield
?ph = 2 & foe = fire
 equipL aether sword
 equipR aether shield
?ph = 2 & foe = ice
 equipL fire sword
 equipR fire shield

// Phase 3: next element on armor gain
?foe.id = dysangelos_perfected &
^foe.armor > lastArmor
 el = (el + 1) % 5
?foe.id = dysangelos_perfected
 lastArmor = foe.armor

// Armor: hammer, else sword
?ph = 3 & foe.armor > 0
 w = elems[el] + " hammer"
:?ph = 3
 w = elems[el] + " sword"

?ph = 3
 s = elems[el] + " shield"
 equipL @w@
 equipR @s@

// No element hammer: heavy hammer
?ph = 3 & foe.armor > 0 &
^item.GetCount(w) = 0
 equip heavy hammer

// Abilities: hammer vs armor, blade
?ph = 3 & foe.armor > 0 &
^item.GetCooldown("heavy_hammer") <= 0
 equip heavy hammer
?ph = 3 & foe.armor = 0 &
^item.GetCooldown("blade") <= 0
 equip blade

// ===== Orb dodge =====
?loc = rocky & dodgeState > 0 &
^foe.state = dodgeState &
^foe.time >= dodgeTime &
^item.GetCooldown("mind") <= 0
 equipL mind

// ===== After boss: potion =====
?foe = boss & loc ! rocky
 bossSeen = 1
?foe.id = dysangelos_perfected
 bossSeen = 1

?bossSeen = 1 & foe ! boss &
^foe.count = 0 & hp < maxhp &
^item.CanActivate()
 activate potion
?bossSeen = 1 & foe ! boss & foe.count = 0
 bossSeen = 0

// ===== Which summon =====
need = 0
wisp = 0
?loc = halls | loc = icy | loc = fungus
 wisp = 1
?wisp = 1 & hasWisp = 0
 sumItem = "fire talisman"
 need = 1
?loc = temple & isBoss = 0 &
^hasVoid = 0
 sumItem = "aether talisman"
 need = 1
?loc = temple & isBoss = 1 &
^tongue = 0 & hasWisp = 0
 sumItem = "fire talisman"
 need = 1

// ===== Summon: must stay last =====
?need = 1 & item.CanActivate() &
^tongue = 0
 equipL @leftW@
 equipR @sumItem@
 activate R
