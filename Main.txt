var heal = 0
var bossSeen = 0
var el = 0
var lastArmor = 0
var ph = 0
var wisp = 0
var isBoss = 0
var ghost = 0
var guard = 0
var poena = 0
var wrap = 0
var hasVoid = 0
var hasWisp = 0
var tongue = 0
var need = 0
var sumItem = ""
var leftW = ""
var w = ""
var s = ""
var elems = [
 "fire", "ice", "aether",
 "poison", "vigor",
]

// Уклонение от шара: числа из замеров
// 0 = уклонение выключено
var dodgeState = 115
var dodgeTime = 64

// Нагараджа: замерить! 0 = выключено
// Замах языком (для отскока)
var nagState = 0
var nagTime = 0
// Язык обвил героя (бить изо всех сил)
var wrapState = 0

// Отскок от молота Стража (замерить)
var guardState = 0
var guardTime = 0

// Дистанция ближнего боя со Стражем
var melee = 8

?loc.begin | loc.loop
 heal = 0
 bossSeen = 0
 el = 0
 lastArmor = 0

isBoss = 0
?foe = boss
 isBoss = 1

ghost = 0
?foe = immune_to_physical
 ghost = 1

poena = 0
?loc = deadwood & isBoss = 1 &
^foe.id = poena
 poena = 1

leftW = "Crusher"
?ghost = 1
 leftW = "fire wand"

// Нагараджа: замах и захват языком
tongue = 0
wrap = 0
?loc = temple & isBoss = 1 &
^nagState > 0 & foe.state = nagState
 tongue = 1
?loc = temple & isBoss = 1 &
^wrapState > 0 & foe.state = wrapState
 wrap = 1
 tongue = 1

// Кто сейчас призван
hasWisp = 0
?summon.count > 0 &
^summon.GetId() = "cinderwisp"
 hasWisp = 1
hasVoid = 0
?summon.count > 0 & hasWisp = 0
 hasVoid = 1

// Отладка: состояния любого босса
?isBoss = 1
 >`0,1,@foe.id@ d=@foe.distance@
 >`0,2,state=@foe.state@ time=@foe.time@
 >`0,3,armor=@foe.armor@ el=@el@
 >`0,4,@foe.buffs.string@

// ===== Общая логика =====
?hp < 10
 activate potion

?hp < 10 & item.potion = empty
 heal = 1

?hp > 40
 heal = 0

?heal = 1
 equipL Crusher
 equipR poison shield
:?foe = boss | foe.count > 5
 equip blade
:
 equipL Crusher
 equipR poison shield

?item.CanActivate("blade")
 activate R
?item.CanActivate("heavy_hammer") &
^poena = 0
 activate R

// ===== Привидения: жезл огня =====
?ghost = 1
 equipL fire wand
 equipR poison shield

// ===== Способности призывов =====
?hasWisp = 1 & foe.count > 0 &
^summon.GetVar("ignition") > 2 &
^item.GetCooldown("cinderwisp") <= 0 &
^item.CanActivate()
 activate cinderwisp

?hasVoid = 1 & foe.count > 0 &
^item.GetCooldown("voidweaver") <= 0 &
^item.CanActivate()
 activate voidweaver

// ===== Храм: отскок от языка =====
?tongue = 1 & wrap = 0 &
^foe.time >= nagTime &
^item.GetCooldown("mind") <= 0
 equipL mind

// ===== Храм: язык обвил, только клинок =====
?wrap = 1
 equip blade

// ===== Шахта: Бронзовый страж =====
guard = 0
?loc = mine & isBoss = 1 & heal = 0
 guard = 1

// Далеко: арбалет
?guard = 1 & foe.distance > melee
 equip aether crossbow

// Вплотную, броня есть: молот
?guard = 1 & foe.distance <= melee &
^foe.armor > 0
 equipL aether hammer
 equipR aether shield

// Вплотную, броня сломана: меч
?guard = 1 & foe.distance <= melee &
^foe.armor = 0
 equipL aether sword
 equipR aether shield

// Способности: тяжёлый молот, клинок
?guard = 1 & foe.distance <= melee &
^foe.armor > 0 &
^item.GetCooldown("heavy_hammer") <= 0
 equip heavy hammer
?guard = 1 & foe.distance <= melee &
^foe.armor = 0 &
^item.GetCooldown("blade") <= 0
 equip blade

// Отскок от удара молотом
?guard = 1 & guardState > 0 &
^foe.state = guardState &
^foe.time >= guardTime &
^item.GetCooldown("mind") <= 0
 equipL mind

// ===== Каньон: Поэна =====
// Тяжёлый молот держит в оглушении
?poena = 1 & heal = 0
 equip heavy hammer
// Клинок готов: один мощный удар
?poena = 1 & heal = 0 &
^item.GetCooldown("blade") <= 0
 equip blade

// ===== Плато: определение фазы =====
ph = 0
?loc = rocky & heal = 0 &
^foe.id = dysangelos_elementalist
 ph = 2
?loc = rocky & heal = 0 &
^foe.id = dysangelos_perfected
 ph = 3

// Фаза 2: контр-стихия
?ph = 2 & foe = poison
 equipL ice sword
 equipR ice shield
?ph = 2 & foe = vigor
 equipL poison sword
 equipR poison shield
?ph = 2 & foe = aether
 equipL vigor sword
 equipR vigor shield
?ph = 2 & foe = fire
 equipL aether sword
 equipR aether shield
?ph = 2 & foe = ice
 equipL fire sword
 equipR fire shield

// Фаза 3: смена стихии при росте брони
?foe.id = dysangelos_perfected &
^foe.armor > lastArmor
 el = (el + 1) % 5
?foe.id = dysangelos_perfected
 lastArmor = foe.armor

// Броня есть: молот, нет: меч
?ph = 3 & foe.armor > 0
 w = elems[el] + " hammer"
:?ph = 3
 w = elems[el] + " sword"

?ph = 3
 s = elems[el] + " shield"
 equipL @w@
 equipR @s@

// Нет молота стихии: тяжёлый молот
?ph = 3 & foe.armor > 0 &
^item.GetCount(w) = 0
 equip heavy hammer

// Способности: молот по броне, клинок
?ph = 3 & foe.armor > 0 &
^item.GetCooldown("heavy_hammer") <= 0
 equip heavy hammer
?ph = 3 & foe.armor = 0 &
^item.GetCooldown("blade") <= 0
 equip blade

// ===== Уклонение от шара =====
?loc = rocky & dodgeState > 0 &
^foe.state = dodgeState &
^foe.time >= dodgeTime &
^item.GetCooldown("mind") <= 0
 equipL mind

// ===== После босса: зелье =====
?foe = boss & loc ! rocky
 bossSeen = 1
?foe.id = dysangelos_perfected
 bossSeen = 1

?bossSeen = 1 & foe ! boss &
^foe.count = 0 & hp < maxhp &
^item.CanActivate()
 activate potion
?bossSeen = 1 & foe ! boss & foe.count = 0
 bossSeen = 0

// ===== Какой призыв нужен =====
need = 0
wisp = 0
?loc = halls | loc = icy | loc = fungus
 wisp = 1
?wisp = 1 & hasWisp = 0
 sumItem = "fire talisman"
 need = 1
?loc = temple & isBoss = 0 &
^hasVoid = 0
 sumItem = "aether talisman"
 need = 1
?loc = temple & isBoss = 1 &
^tongue = 0 & hasWisp = 0
 sumItem = "fire talisman"
 need = 1

// ===== Вызов: строго в самом конце =====
?need = 1 & item.CanActivate() &
^tongue = 0
 equipL @leftW@
 equipR @sumItem@
 activate R
